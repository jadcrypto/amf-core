def _parse_header(self, data):
    magic = struct.unpack("<I", data[:4])[0]
    if magic == 0x46554747: # إذا وصل مقلوباً
        self.endian = ">" # Big-endian
    elif magic == 0x46475547:
        self.endian = "<" # Little-endian
    else:
        raise ValueError("File format not recognized as GGUF")
